Candidate Details

' + fields.map(([label, id]) => '' ).join('') + '
' + esc(label) + '' + esc(get(id)) + '
'; result.classList.remove('hidden'); score.classList.add('hidden'); error.classList.add('hidden'); setField('ldceName', get('candidateName')); setField('ldceZone', get('zone')); setField('ldcePost', get('examPost')); if (!url) { error.textContent = 'कृपया Response Sheet URL भरें।'; error.classList.remove('hidden'); return; } if (!/^https?:\/\//i.test(url)) { error.textContent = 'Response Sheet URL सही नहीं है।'; error.classList.remove('hidden'); return; } score.innerHTML = '

RAILWAY RANK MITRA — SCORE CARD

' + '

Response Sheet से डेटा पढ़ा जा रहा है...

'; score.classList.remove('hidden'); try { const response = await fetch(url, { method: 'GET', mode: 'cors', credentials: 'omit', redirect: 'follow' }); if (!response.ok) { throw new Error('RRB सर्वर ने HTTP ' + response.status + ' दिया। प्रश्न-पत्र पढ़ा नहीं जा सका।'); } const html = await response.text(); const doc = new DOMParser().parseFromString(html, 'text/html'); if (!doc.body || !doc.body.innerText.trim()) { throw new Error('Response Sheet खाली है या पढ़ी नहीं जा सकती।'); } const pageText = doc.body.innerText; const normalizedText = pageText.toLowerCase(); /* अलग-अलग वेबसाइटों पर Response Sheet का प्रारूप अलग होता है। प्रश्न, चुने गए उत्तर और सही उत्तर के स्पष्ट संकेत ढूँढ़े जाते हैं। */ const hasQuestions = /question\s*(id|no|number)|प्रश्न\s*(संख्या|नं|आईडी)/i.test(pageText); const hasCandidateAnswers = /chosen option|candidate answer|your answer|selected answer|आपका उत्तर/i .test(pageText); const hasCorrectAnswers = /correct option|correct answer|official answer|सही उत्तर/i .test(pageText); if (!hasQuestions || !hasCandidateAnswers || !hasCorrectAnswers) { throw new Error( 'URL खुल गया, लेकिन इसमें प्रश्न ID, उम्मीदवार के उत्तर और ' + 'आधिकारिक सही उत्तर का पूरा डेटा पहचान में नहीं आया। ' + 'इसलिए वास्तविक स्कोर निकालना संभव नहीं हुआ।' ); } /* सिर्फ पेज में शब्द मिल जाने से उत्तरों का सही मिलान प्रमाणित नहीं होता। प्रश्नों की संरचना और प्रत्येक प्रश्न की Answer Key का मिलान जरूरी है। */ score.innerHTML = '

RAILWAY RANK MITRA — SCORE CARD

' + '

Candidate: ' + esc(get('candidateName')) + '

' + '

Registration Number: ' + esc(get('regNo')) + '

' + '
' + 'Response Sheet का पेज पढ़ लिया गया है, लेकिन इस प्रारूप के लिए ' + 'प्रश्नवार उत्तर निकालने और आधिकारिक Answer Key से उनका मिलान ' + 'करने वाला parser अभी सत्यापित नहीं है। इसलिए सही, गलत और कुल अंक ' + 'दिखाना अभी संभव नहीं है।' + '
'; } catch (err) { score.classList.add('hidden'); error.textContent = 'Score Card नहीं बन पाया: ' + err.message + ' यदि CORS या 403 की समस्या है, तो URL पढ़ने के लिए सर्वर-साइड ' + 'व्यवस्था की आवश्यकता होगी।'; error.classList.remove('hidden'); } result.scrollIntoView({ behavior: 'smooth', block: 'start' }); });

document.getElementById("candidateForm").addEventListener("submit", async function(e) { e.preventDefault(); const name = document.getElementById("candidateName").value.trim(); const regNo = document.getElementById("regNo").value.trim(); const examDate = document.getElementById("examDate").value; const responseURL = document.getElementById("responseURL").value.trim(); const details = document.getElementById("candidateResult"); const score = document.getElementById("rrbScoreResult"); const error = document.getElementById("rrbScoreError"); details.innerHTML = ""; score.innerHTML = ""; error.innerHTML = ""; details.innerHTML = `

Candidate Details

Candidate Name: ${esc(name)}

Registration Number: ${esc(regNo)}

Exam Date: ${esc(examDate)}

Exam Type: ${esc(document.getElementById("examType").value)}

Post: ${esc(document.getElementById("examPost").value)}

Zone: ${esc(document.getElementById("zone").value)}

Shift: ${esc(document.getElementById("shift").value)}

Category: ${esc(document.getElementById("category").value)}

`; if (!responseURL) { error.innerHTML = `
Response Sheet URL भरें। बिना प्रश्नों के डेटा के स्कोर नहीं निकाला जा सकता।
`; return; } score.innerHTML = `
Response Sheet का डेटा पढ़ा जा रहा है...
`; try { const response = await fetch(responseURL); if (!response.ok) { throw new Error("Response Sheet URL नहीं खुला।"); } const html = await response.text(); const doc = new DOMParser().parseFromString(html, "text/html"); const tables = [...doc.querySelectorAll("table")]; let questions = []; for (const table of tables) { const rows = [...table.querySelectorAll("tr")]; if (rows.length < 2) continue; const headers = [...rows[0].querySelectorAll("th,td")] .map(cell => cell.innerText.trim().toLowerCase()); const chosenIndex = headers.findIndex(h => /chosen option|your answer|candidate answer|selected answer|response/.test(h) ); const correctIndex = headers.findIndex(h => /correct option|correct answer|official answer/.test(h) ); if (chosenIndex < 0 || correctIndex < 0) continue; questions = rows.slice(1).map(row => { const cells = [...row.querySelectorAll("td")] .map(cell => cell.innerText.trim()); return { chosen: cells[chosenIndex] || "", correct: cells[correctIndex] || "" }; }).filter(q => q.chosen || q.correct); if (questions.length) break; } if (!questions.length) { throw new Error( "URL खुल गया, लेकिन Candidate Response और Correct Answer वाला प्रश्न-डेटा नहीं मिला।" ); } const isBlank = value => !value || /^(--|na|n\/a|not answered|not attempted|-)$/.test(value.trim().toLowerCase()); const normalize = value => value.trim().toLowerCase().replace(/\s+/g, " "); let attempted = 0; let correct = 0; let wrong = 0; let unattempted = 0; questions.forEach(q => { if (isBlank(q.chosen)) { unattempted++; } else { attempted++; if (!isBlank(q.correct) && normalize(q.chosen) === normalize(q.correct)) { correct++; } else { wrong++; } } }); score.innerHTML = `

📊 Candidate Score Board

Total Questions: ${questions.length}

Attempted: ${attempted}

Correct: ${correct}

Wrong: ${wrong}

Unattempted: ${unattempted}

Total Score: आधिकारिक मार्किंग स्कीम की पुष्टि आवश्यक

`; } catch (err) { score.innerHTML = ""; error.innerHTML = `
Score Board नहीं निकल पाया।
${esc(err.message)}

Railway वेबसाइट का URL बाहरी वेबसाइट से पढ़ने की अनुमति नहीं देता हो सकता है। ऐसी स्थिति में सर्वर की मदद से डेटा पढ़ने की सुविधा बनानी होगी।
`; } });
